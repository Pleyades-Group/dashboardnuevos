Conexión Google Apps Script Web App

Conecta directamente tu hoja de Google Sheets en tiempo real

Copia este código en tu archivo de Extensiones > Apps Script en Google Sheets y publícalo como Web App (Acceso: Cualquier persona / Anyone) para enviar la fecha exacta de modificación: 

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("U123 Detalle") || ss.getSheets()[0];
    var rawValues = sheet.getDataRange().getValues();
    var processedData = [];
    var timeZone = ss.getSpreadsheetTimeZone() || "GMT-6";
    
    for (var i = 0; i < rawValues.length; i++) {
      var row = [];
      var hasData = false;
      for (var j = 0; j < rawValues[i].length; j++) {
        var val = rawValues[i][j];
        if (val instanceof Date) {
          val = Utilities.formatDate(val, timeZone, "yyyy-MM-dd");
        }
        row.push(val);
        if (val !== "" && val !== null) hasData = true;
      }
      if (hasData) processedData.push(row);
    }
    
    var lastUpdated = ss.getLastUpdated();
    var formattedDate = Utilities.formatDate(lastUpdated, timeZone, "yyyy-MM-dd HH:mm:ss");
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      lastUpdated: formattedDate,
      data: processedData
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
